+#ifndef SWEET_FB_COHERENT_H
+#define SWEET_FB_COHERENT_H
+/*
+ * 0394 (fbcoh-20261112): coherent CPU copy of the swap layers, pure and host-testable (no IOKit, no registers).
+ *
+ * Why (live capture q10jfh, Photos grid scroll): src_race +91, busy_after +39 in ~600 pushes, only while scrolling.  The detached
+ * layer 0 is the APP's own drawable; SwapWait only paces backboardd, so the app's GPU starts rendering its next frame into a
+ * surface the kernel is still copying (mixed frames, tile blocks).
+ *
+ * Rule: a layer copy is good when no GPU write to any of the frame's surfaces was queued, running or promised (Nord2GPU
+ * surfaceBusy: write jobs + 0112 reservations) from before the copy started until after it ended, and the sampled rows did not
+ * change.  The copy checks busy every SWEET_FB_COH_CHUNK_ROWS rows; on a hit (or a hash change, or busy right after) it waits
+ * until the surfaces are idle and copies the frame again (the newer, complete frame).  Waiting is bounded by
+ * SWEET_FB_COH_BUDGET_US per present and SWEET_FB_COH_MAX_ATTEMPTS copies.  Then, if the copy went into the hidden panel
+ * buffer (double buffering) and fewer than SWEET_FB_COH_MAX_SKIPS presents in a row were skipped, the present is skipped (the
+ * panel keeps the last good frame; the next swap is copied normally); otherwise the last copy is shown as before 0394 (and the
+ * last copy then runs to its end without chunk checks, so it is never half written).
+ */
+#include <stdbool.h>
+#include <stdint.h>
+
+#define SWEET_FB_COH_CHUNK_ROWS     128u       /* busy check granularity: ~0.15 ms of copy */
+#define SWEET_FB_COH_BUDGET_US      8000u      /* total wait for idle surfaces per present */
+#define SWEET_FB_COH_MAX_ATTEMPTS   3u         /* copies per present */
+#define SWEET_FB_COH_MAX_SKIPS      2u         /* consecutive presents skipped (old frame kept) before an unstable copy is shown */
+
+enum sweet_fb_coh_verdict {
+	SWEET_FB_COH_SHOW = 0,          /* the copy is coherent: show it */
+	SWEET_FB_COH_RETRY,             /* wait for idle surfaces, copy again */
+	SWEET_FB_COH_GIVE_UP,           /* budget or attempts exhausted: show the last copy (counted) */
+	SWEET_FB_COH_SKIP,              /* exhausted and the copy went to the hidden buffer: keep the shown frame, present nothing new */
+};
+
+/* After copy number `attempt` (0-based).  aborted: busy seen between chunks; raced: sampled rows changed; busy_after: a write is
+ * pending right after the copy; waited_us: wait spent so far in this present. */
+static inline int
+sweet_fb_coh_verdict(bool enabled, unsigned attempt, bool aborted, bool raced, bool busy_after, uint64_t waited_us, bool skip_ok)
+{
+	if (!enabled || (!aborted && !raced && !busy_after)) {
+		return SWEET_FB_COH_SHOW;
+	}
+	if (attempt + 1 >= SWEET_FB_COH_MAX_ATTEMPTS || waited_us >= SWEET_FB_COH_BUDGET_US) {
+		return skip_ok ? SWEET_FB_COH_SKIP : SWEET_FB_COH_GIVE_UP;
+	}
+	return SWEET_FB_COH_RETRY;
+}
+
+/* A skip is allowed when the copy target is not the buffer the overlay shows and the skip run is short. */
+static inline bool
+sweet_fb_coh_skip_ok(bool double_buffered, bool target_hidden, uint32_t skip_run)
+{
+	return double_buffered && target_hidden && skip_run < SWEET_FB_COH_MAX_SKIPS;
+}
+
+/* Chunk checks are useful while another copy can follow, or when an unstable copy would be skipped anyway. */
+static inline bool
+sweet_fb_coh_chunk_checks(bool enabled, unsigned attempt, uint64_t waited_us, bool skip_ok)
+{
+	return enabled && ((attempt + 1 < SWEET_FB_COH_MAX_ATTEMPTS && waited_us < SWEET_FB_COH_BUDGET_US) || skip_ok);
+}
+
+/* Busy check due before copying row index r (0-based within the copy loop): every chunk boundary except the first row (the
+ * caller checks once before the copy starts). */
+static inline bool
+sweet_fb_coh_check_due(uint32_t r)
+{
+	return r != 0 && (r % SWEET_FB_COH_CHUNK_ROWS) == 0;
+}
+
+/* How long the next idle poll may sleep: 1 ms sleeps while more than 1.2 ms of budget remain, else 100 us spins. 0 = budget over. */
+static inline uint32_t
+sweet_fb_coh_poll_us(uint64_t waited_us)
+{
+	if (waited_us >= SWEET_FB_COH_BUDGET_US) {
+		return 0;
+	}
+	return (SWEET_FB_COH_BUDGET_US - waited_us > 1200u) ? 1000u : 100u;
+}
+
+struct sweet_fb_coh_stats {
+	uint64_t presents, prewait, aborts, raced, busy_after, retries, recovered, gave_up, skipped, wait_us_sum, wait_us_max;
+};
+
+#endif
